# Testing a package walks its submodules. Each package's manifest.py is a firmware build
# manifest and appkit.hostkeys needs a board's USB host (usb.core over _usbhost), so
# neither imports under CPython here; mypy checks their stubs.

appkit\.hostkeys
# songmodel.link imports where hmilink is installed, and not elsewhere. Its stub names
# hmilink's Model base and the Host protocol for type checkers only; neither is a name
# the module itself defines.
songmodel\.link
songmodel\.link\.(Host|Model)
